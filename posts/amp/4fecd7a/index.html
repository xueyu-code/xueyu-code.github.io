龙芯LoongArch异构多核处理器(AMP)框架详解

框架概述

这是一个基于龙芯LoongArch架构的**异构多核处理器(AMP - Asymmetric Multi-Processing)**框架，实现了Linux系统与裸机程序在双核系统中的协作处理。该框架允许CPU0运行完整的Linux系统，而CPU1运行高性能的rewors程序(理论上也可以是别的程序，比如vxworks的，只要BSP做好即可，毕竟CPU不会去区分到底是什么程序)，两者通过共享内存和核间中断进行通信(上层设计为虚拟网卡)。

核心特性

  • 异构处理：CPU0运行Linux，CPU1运行裸机程序
  • 零拷贝通信：基于共享内存的高效数据传输
  • 实时性保证：高实时RTOS程序提供确定性实时响应
  • 标准接口：提供标准的网络设备接口
  • 可扩展性：支持多种RTOS和Linux应用的集成
  • reworks终端对应的串口中断以及2个CPU的2个IPI中断已经在BSP中做好中断路由
  • 动态控制：支持运行时启停CPU核心，该框架运行之后如果用户想让CPU1回到Linux调度, 可以让用户在reworks的终端中退出系统，此时即可让出CPU。然后再通过Linux下的CPU热插拔机制 重新加载CPU1，此时即可回到Linux调度。

static void xy_csr_mail_send(uint64_t data, int cpu, int mailbox)
{
    uint64_t val;
    // 发送高32位
    val = IOCSR_MBUF_SEND_BLOCKING;
    val |= (IOCSR_MBUF_SEND_BOX_HI(mailbox) << IOCSR_MBUF_SEND_BOX_SHIFT);
    val |= (cpu << IOCSR_MBUF_SEND_CPU_SHIFT);
    val |= (data & IOCSR_MBUF_SEND_H32_MASK);
    iocsr_write64(val, LOONGARCH_IOCSR_MBUF_SEND);
    // 发送低32位
    val = IOCSR_MBUF_SEND_BLOCKING;
    val |= (IOCSR_MBUF_SEND_BOX_LO(mailbox) << IOCSR_MBUF_SEND_BOX_SHIFT);
    val |= (cpu << IOCSR_MBUF_SEND_CPU_SHIFT);
    val |= (data << IOCSR_MBUF_SEND_BUF_SHIFT);
    iocsr_write64(val, LOONGARCH_IOCSR_MBUF_SEND);
}

static long cpu_handle(unsigned int cmd, unsigned long arg)
{
    switch(cmd) {
        case STOP:  
            cpu_down(1);                    // 停止CPU1
            printk(KERN_INFO "stop cpu1");
            break;
        case START:
            printk(KERN_INFO "start cpu1");
            xy_csr_mail_send(entry, 1, 0); // 发送程序入口地址
            send_ipi_single_func(1, 1);     // 发送核间中断唤醒CPU1
            break;
    }
}

void load_elf_to_preallocated_memory(const char *filename) 
{
    // 第一次遍历：计算地址范围
    for (section = abfd->sections; section != NULL; section = section->next) {
        bfd_vma vma = bfd_section_vma(section);
        size = bfd_section_size(section);
        
        if (vma < lowest_vma) lowest_vma = vma;
        if (vma + size > highest_vma) highest_vma = vma + size;
    }
    // 第二次遍历：加载段内容
    for (section = abfd->sections; section != NULL; section = section->next) {
        bfd_vma offset = vma - lowest_vma;
        
        if ((bfd_section_flags(section) & SEC_LOAD) != 0) {
            // 加载可执行段
            bfd_get_section_contents(abfd, section, preallocated_memory + offset, 0, size);
        } else if (!(bfd_section_flags(section) & SEC_HAS_CONTENTS)) {
            // 清零BSS段
            memset(preallocated_memory + offset, 0, size);
        }
        // 刷新CPU缓存
        __builtin___clear_cache((void*)(preallocated_memory + offset), 
                               (void*)(preallocated_memory + offset + size));
    }
}

unsigned long get_symbol_address(const char *symbol_name)
{
    FILE *fp = fopen("/proc/kallsyms", "r");
    
    while (fgets(line, sizeof(line), fp)) {
        token = strtok(line, " ");           // 地址
        address = strtoul(token, NULL, 16);  // 转换为数值
        
        token = strtok(NULL, " ");           // 符号类型
        token = strtok(NULL, " \n");         // 符号名称
        
        if (strcmp(token, symbol_name) == 0) {
            return address;
        }
    }
    return 0;
}

static netdev_tx_t my_start_xmit(struct sk_buff *skb, struct net_device *dev)
{
    struct my_netdev_priv *priv = netdev_priv(dev);

retry:
    // 写入TX环形缓冲区
    if (write_tx_ring(priv->shm_id, skb->data, skb->len) != 0) {
        printk("my_start_xmit write tx failed!.....\n");
        goto retry;  // 缓冲区满时重试
    }

    // 更新统计信息
    priv->stats.tx_packets++;
    priv->stats.tx_bytes += skb->len;
    dev_kfree_skb(skb);

    // 通知对方CPU有新数据
    if (!is_rxing(priv->shm_id)) {
        send_my_rx_ipi(1); // 发送核间中断
    }

    return NETDEV_TX_OK;
}

int write_tx_ring(s_shm_id_t id, void* data, shm_u32_t size)
{
    s_shm_t* shm = s_shms + id;
    s_buffer_t* buffer = NULL;
    s_ring_t* ring = (shm->role == ROLE_HOST) ? &(shm->tx_ring) : &(shm->rx_ring);

    buffer = ring->current_pos;

    // 原子检查缓冲区状态
    if (__atomic_load_n(&buffer->flag, __ATOMIC_SEQ_CST) != FLAG_CLEAN) {
        return -1;  // 缓冲区忙，返回失败
    }

    // 复制数据到缓冲区
    memcpy(buffer->data, data, size);
    buffer->payload_size = size;

    // 原子设置缓冲区为脏状态
    __atomic_store_n(&buffer->flag, FLAG_DIRTY, __ATOMIC_SEQ_CST);

    // 移动到下一个缓冲区
    if ((s_shm_addr_t)(buffer + 1) > ring->end) {
        ring->current_pos = (s_buffer_t*)(ring->start);  // 环形回绕
    } else {
        ring->current_pos = buffer + 1;
    }

    return 0;
}

static void my_rx(void* dev)
{
    struct my_netdev_priv *priv = netdev_priv((struct net_device*)dev);
    unsigned int size;

    do {
        struct sk_buff *skb;
retry:
        // 从RX环形缓冲区读取数据
        if (read_rx_ring(priv->shm_id, (void**)&skb, &size) != 0) {
            up_rxing(priv->shm_id, 0);           // 标记非接收状态
            wait_for_completion(&my_completion);  // 等待中断唤醒
            up_rxing(priv->shm_id, 1);           // 标记接收状态
            reinit_completion(&my_completion);
            goto retry;
        }

        // 设置网络包属性
        skb->dev = dev;
        skb->protocol = eth_type_trans(skb, dev);
        skb->ip_summed = CHECKSUM_UNNECESSARY;

        // 更新统计信息
        priv->stats.rx_packets++;
        priv->stats.rx_bytes += size;

        // 提交到网络栈
        netif_rx_ni(skb);
    } while(1);
}

int read_rx_ring(s_shm_id_t id, void** data, shm_u32_t* size)
{
    s_shm_t* shm = s_shms + id;
    s_buffer_t* buffer = NULL;
    s_ring_t* ring = (shm->role == ROLE_HOST) ? &(shm->rx_ring) : &(shm->tx_ring);
    struct sk_buff** skb = (struct sk_buff**)data;

    buffer = ring->current_pos;

    // 原子检查是否有数据
    if (__atomic_load_n(&buffer->flag, __ATOMIC_SEQ_CST) != FLAG_DIRTY) {
        return -1;  // 没有数据
    }

    *size = buffer->payload_size;

    // 分配sk_buff
    *skb = dev_alloc_skb(*size);
    if (!(*skb)) {
        printk("dev_alloc_skb failed!, size = %d\n", *size);
        return -2;
    }

    // 复制数据到sk_buff
    memcpy(skb_put(*skb, *size), (char*)(buffer->data), *size);

    // 原子清除缓冲区状态
    __atomic_store_n(&buffer->flag, FLAG_CLEAN, __ATOMIC_SEQ_CST);

    // 移动到下一个缓冲区
    if ((s_shm_addr_t)(buffer + 1) > ring->end) {
        ring->current_pos = (s_buffer_t*)(ring->start);
    } else {
        ring->current_pos = buffer + 1;
    }

    return 0;
}

// 缓冲区结构（64字节对齐优化缓存性能）
typedef struct __attribute__((aligned(64))) s_buffer {
    shm_u8_t data[BUFFER_DATA_MAX_SIZE];  // 数据载荷
    shm_u32_t flag;                       // 状态标志：CLEAN/DIRTY
    shm_u32_t payload_size;               // 有效数据大小
    shm_u32_t is_rxing;                   // 接收状态标志
    shm_u32_t reserve;                    // 保留字段
} s_buffer_t;
// 环形队列管理结构
typedef struct __attribute__((aligned(64))) s_ring {
    s_shm_addr_t start;        // 起始地址
    s_shm_addr_t end;          // 结束地址
    s_buffer_t* current_pos;   // 当前位置指针
} s_ring_t;
// 共享内存描述符
typedef struct __attribute__((aligned(64))) s_shm {
    shm_role_t role;           // 角色：HOST/REMOTE
    s_shm_addr_t va;           // 虚拟地址
    s_shm_addr_t pa;           // 物理地址
    s_shm_size_t size;         // 内存大小
    s_shm_id_t id;             // 共享内存ID
    s_ring_t tx_ring;          // 发送环形缓冲区
    s_ring_t rx_ring;          // 接收环形缓冲区
} s_shm_t;

static netdev_tx_t my_start_xmit(struct sk_buff *skb, struct net_device *dev)
{
    struct my_netdev_priv *priv = netdev_priv(dev);
retry:
    // 写入TX环形缓冲区
    if (write_tx_ring(priv->shm_id, skb->data, skb->len) != 0) {
        printk("my_start_xmit write tx failed!.....\n");
        goto retry;  // 缓冲区满时重试
    }
    // 更新统计信息
    priv->stats.tx_packets++;
    priv->stats.tx_bytes += skb->len;
    dev_kfree_skb(skb);

    // 通知对方CPU有新数据
    if (!is_rxing(priv->shm_id)) {
        send_my_rx_ipi(1); // 发送核间中断
    }

    return NETDEV_TX_OK;
}

static void my_rx(void* dev)
{
    struct my_netdev_priv *priv = netdev_priv((struct net_device*)dev);
    
    do {
        struct sk_buff *skb;
retry:
        // 从RX环形缓冲区读取数据
        if (read_rx_ring(priv->shm_id, (void**)&skb, &size) != 0) {
            up_rxing(priv->shm_id, 0);           // 标记非接收状态
            wait_for_completion(&my_completion);  // 等待中断唤醒
            up_rxing(priv->shm_id, 1);           // 标记接收状态
            reinit_completion(&my_completion);
            goto retry;
        }
        // 设置网络包属性
        skb->dev = dev;
        skb->protocol = eth_type_trans(skb, dev);
        skb->ip_summed = CHECKSUM_UNNECESSARY;
        // 更新统计信息
        priv->stats.rx_packets++;
        priv->stats.rx_bytes += size;
        // 提交到内核网络协议栈
        netif_rx_ni(skb);
    } while(1);
}

性能优化特性

  • 共享内存：避免数据在内核空间和用户空间之间拷贝
  • ring buf设计：无锁环形缓冲区设计
  • 原子操作：使用__atomic_load_n和__atomic_store_n保证多核并发安全

存在问题

  • 由于时间仓促同时也为了便于用户使用，不得不走了一遍内核网络协议栈，这部分是性能瓶颈
  • 由于2K1000的phy1,phy2都是挂在龙芯内部同一个PCIE控制器上，所以无法做到把2个phy在OS层 的隔离控制，不得不使用网桥，这部分也是性能瓶颈
  • 高速触发IPI中断时，有时候会丢中断，联系龙芯官方得知无法优化，当时芯片设计的时候没有考虑到AMP的需求，所以这部分无法优化，他们官方测也是会偶发性丢中断
  • iperf3测试时某些情况时会出现系统崩溃的情况，详情参考专门的测试文档
